% ---------- ATS-safe encoding and typography ----------
% f-ligatures (fi, fl, ff) are single glyphs that many ATS parsers extract as
% U+FB01 or drop entirely, mangling words like "aficionado" and "profile".
% Both engine paths below disable them and keep the text layer plain ASCII.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{lmodern}
  \input{glyphtounicode}
  \pdfgentounicode=1
  \usepackage{microtype}
  \DisableLigatures[f]{encoding = *, family = *}
\else
  \usepackage{fontspec}
  % Referenced by file name so the build works without fontconfig (CI/Tectonic).
  \setmainfont{lmroman10-regular.otf}[
    BoldFont       = lmroman10-bold.otf,
    ItalicFont     = lmroman10-italic.otf,
    BoldItalicFont = lmroman10-bolditalic.otf,
    Ligatures      = NoCommon
  ]
  \usepackage{microtype}
\fi
